<clinical_trial>
    <title>Multicenter Phase II Study Evaluating the Efficacy of Cetuximab in Combination with Personalized Chemotherapy Regimens for Treatment of Advanced Gastric Cancer</title>
    <design>
        <phase>II</phase>
        <objective>The primary objective of this study is to evaluate the overall response rate of cetuximab in conjunction with personalized chemotherapy regimens (including oxaliplatin and capecitabine) in patients with advanced gastric cancer, with attention to mechanistic synergy and optimized therapeutic regimens.</objective>
        <endpoints>
            <primary>Overall response rate (ORR) according to RECIST criteria</primary>
            <secondary>
                <item>Progression-free survival (PFS)</item>
                <item>Overall survival (OS)</item>
                <item>Toxicity assessment using CTCAE v5.0</item>
            </secondary>
        </endpoints>
    </design>
    <population>
        <description>The trial aims to recruit 100 patients diagnosed with histologically confirmed advanced gastric cancer, unresponsive to first-line therapy, to assess the benefits of cetuximab in a cohort with specific molecular profiles.</description>
        <inclusion_criteria>
            <item>Histologically confirmed adenocarcinoma of the stomach</item>
            <item>Unresectable or metastatic gastric cancer</item>
            <item>No previous treatments targeting the EGFR pathway</item>
            <item>Eastern Cooperative Oncology Group (ECOG) performance status ≤ 2</item>
            <item>Adequate hematological, renal, and hepatic function</item>
            <item>Age ≥ 18 years and able to provide informed consent</item>
        </inclusion_criteria>
        <exclusion_criteria>
            <item>Presence of active central nervous system metastases</item>
            <item>Histologically mixed gastric cancer or variants with signet ring morphology</item>
            <item>Serious comorbidities that could interfere with study assessments</item>
            <item>Pregnant or breastfeeding women</item>
            <item>Previous malignancies (except adequately treated non-melanoma skin cancer or in-situ carcinoma)</item>
        </exclusion_criteria>
    </population>
    <treatment>
        <description>Patients will receive cetuximab at a loading dose of 400 mg/m² followed by a weekly maintenance dose of 250 mg/m². Concurrent with cetuximab, patients will receive personalized chemotherapy based on molecular profiling, including oxaliplatin (130 mg/m² every 3 weeks) and capecitabine (1000 mg/m² twice daily for 14 days of a 21-day cycle).</description>
        <duration>The treatment duration for each patient will be up to 6 cycles or until disease progression, unacceptable toxicity, or withdrawal of consent.</duration>
    </treatment>
    <monitoring>
        <description>Close monitoring for adverse events and therapeutic responses will be instituted at every cycle, with imaging assessments conducted every 8 weeks to evaluate tumor response.</description>
        <adaptive_strategies>Any significant adverse events or lack of treatment response will prompt timely adjustments to the chemotherapy regimen to enhance patient safety and treatment outcomes.</adaptive_strategies>
    </monitoring>
    <date>
        <start_date>January 2023</start_date>
        <anticipated_completion_date>December 2025</anticipated_completion_date>
    </date>
</clinical_trial>
