<ClinicalTrial>
    <Title>A Randomized, Open-label, Multi-centre Study Evaluating the Effect of Insulin Degludec on Quality of Life in Patients With Type 2 Diabetes Mellitus</Title>
    <StudyType>Randomized, Open-label, Multi-centre</StudyType>
    <Location>Multi-national (Asia, Europe, North America)</Location>
    <Aim>The objective of this study is to evaluate the impact of insulin degludec on quality of life, glycaemic control, and patient-reported outcomes in individuals with type 2 diabetes mellitus over a 24-week period.</Aim>
    
    <InclusionCriteria>
        <Criterion>Patients aged 18 years or older diagnosed with type 2 diabetes mellitus.</Criterion>
        <Criterion>Patients currently receiving oral antidiabetic medications or basal insulin.</Criterion>
        <Criterion>Patients with HbA1c levels between 7.0% and 10.0% at baseline.</Criterion>
        <Criterion>Patients who have provided informed consent to participate in the study.</Criterion>
    </InclusionCriteria>

    <ExclusionCriteria>
        <Criterion>History of severe hypoglycaemia requiring medical intervention in the past 6 months.</Criterion>
        <Criterion>Pregnant or breastfeeding women, or women planning to conceive within the study duration.</Criterion>
        <Criterion>Participation in any investigational drug study within the last 3 months.</Criterion>
        <Criterion>Uncontrolled endocrine disorder or a significant history of other comorbidities impacting diabetes management.</Criterion>
    </ExclusionCriteria>

    <Endpoints>
        <PrimaryEndpoint>Change in Diabetes Quality of Life (DQOL) score from baseline to week 24.</PrimaryEndpoint>
        <SecondaryEndpoints>
            <Endpoint>Change in HbA1c levels from baseline to week 24.</Endpoint>
            <Endpoint>Incidence of adverse events (AEs) during the study period.</Endpoint>
            <Endpoint>Change in patient-reported outcomes measured via validated questionnaires.</Endpoint>
        </SecondaryEndpoints>
    </Endpoints>

    <StudyDesign>
        <Randomization>Patients will be randomly assigned in a 1:1 ratio to receive either insulin degludec or standard treatment (insulin glargine).</Randomization>
        <Duration>The study will involve a 24-week treatment phase, followed by a 4-week follow-up period.</Duration>
        <FollowUp>Patients will be monitored for glycaemic control and quality of life metrics at baseline, week 12, and week 24.</FollowUp>
    </StudyDesign>

    <StatisticalAnalysis>
        <AnalysisPlan>Descriptive statistics will be utilized to summarize baseline characteristics. Changes from baseline to endpoint will be analyzed using paired t-tests or Wilcoxon signed-rank tests, as appropriate. A p-value of <0.05 will be considered statistically significant.</AnalysisPlan>
    </StatisticalAnalysis>

    <ExpectedOutcomes>
        <Outcome>Improvement in quality of life scores, reflecting superior patient satisfaction and management of diabetes.</Outcome>
        <Outcome>Reduction in HbA1c levels indicating effective glycaemic control with insulin degludec.</Outcome>
        <Outcome>A favorable safety profile with a low incidence of hypoglycaemia and other adverse events.</Outcome>
    </ExpectedOutcomes>
</ClinicalTrial>
